## Estrutura de Monetização (3 Camadas)

### Camada 1: Ads (Free Users)
```
Usuário Free vê ads a cada:
- Final de cada fase (5 seg)
- Morte do personagem (3 seg)
- Leaderboard refresh (banner)

Renda:
- 10k Free users × 2 ads/dia × R$ 0,002/ad
= R$ 40/dia = R$ 1.200/mês

Isso não chega em nada MAS:
- Oferece monetização sem forçar compra
- Users não saem por ads demais
```

### Camada 2: In-App Purchases (Free Users)
```
Cosmetics:
- Skins de personagem: R$ 4,99 cada
- Efeitos especiais: R$ 2,99
- Themes leaderboard: R$ 1,99

Power-ups temporiários:
- 1h de poder especial: R$ 0,99
- 10x pontos: R$ 1,99

Renda estimada:
- 10k Free users
- 2% compram cosmetics (200 users)
- Ticket médio: R$ 8
= R$ 1.600/mês (apenas cosmetics)
```

### Camada 3: Subscription Pro (R$ 9,99/mês)
```
10k Free users
20% convertem para Pro (2.000 users)
2.000 × R$ 9,99 = R$ 19.980/mês

Muito melhor! 🚀
```

## Código: Sistema Freemium

### Frontend (React)

```typescript
// src/games/FreemiumGate.tsx
import React from 'react';

interface FreemiumProps {
  timePlayedMinutes: number;
  showMonetization: boolean;
}

export const FreemiumGate: React.FC<FreemiumProps> = ({
  timePlayedMinutes,
  showMonetization
}) => {
  const isPro = localStorage.getItem('isPro') === 'true';

  // Após 60 minutos, mostrar upgrade offer
  if (timePlayedMinutes > 60 && !isPro && showMonetization) {
    return (
      <div className="monetization-modal">
        <h2>Você é incrível! 🎮</h2>
        <p>Desbloqueie 50+ fases com PRO</p>
        
        <div className="upgrade-options">
          {/* Opção 1: Cosmetics */}
          <div className="cosmetic-shop">
            <h3>Customizar Personagem</h3>
            <button onClick={() => buyCosmeticSkin('ninja', 4.99)}>
              Skin Ninja - R$ 4,99
            </button>
            <button onClick={() => buyCosmeticSkin('gold', 2.99)}>
              Efeito Ouro - R$ 2,99
            </button>
          </div>

          {/* Opção 2: PRO Subscription */}
          <div className="pro-subscription">
            <h3>🌟 Upgrade para PRO</h3>
            <p>R$ 9,99/mês</p>
            <ul>
              <li>✅ 50+ fases desbloqueadas</li>
              <li>✅ Sem ads</li>
              <li>✅ Cosmetics grátis</li>
              <li>✅ 1h ONZAP Premium/mês</li>
            </ul>
            <button onClick={upgradeToPro}>
              Assinar PRO Agora
            </button>
          </div>

          {/* Opção 3: Continue Free */}
          <button onClick={closeMontization}>
            Continuar Jogando (com ads)
          </button>
        </div>
      </div>
    );
  }

  // Render game normally
  return <GameComponent />;
};

// Backend (NestJS)
@Controller('api/game/purchase')
export class GamePurchaseController {
  constructor(private stripe: StripeService) {}

  @Post('cosmetic')
  async buyCosmeticSkin(@Body() dto: BuyDto, @Req() req) {
    const userId = req.user.id;
    
    // Stripe payment
    const payment = await this.stripe.charge({
      customerId: userId,
      amount: dto.price * 100, // cents
      description: `Cosmetic: ${dto.itemId}`
    });

    if (payment.success) {
      // Save cosmetic to user
      await this.prisma.userCosmetic.create({
        data: {
          userId,
          cosmetic: dto.itemId,
          purchasedAt: new Date()
        }
      });

      // Analytics
      await this.analytics.track({
        event: 'cosmetic_purchase',
        userId,
        amount: dto.price,
        itemId: dto.itemId
      });

      return { success: true, cosmetic: dto.itemId };
    }
  }

  @Post('subscribe-pro')
  async subscribeToPro(@Body() dto: SubscribeDto, @Req() req) {
    const userId = req.user.id;

    // Create Stripe subscription
    const subscription = await this.stripe.subscribe({
      customerId: userId,
      priceId: process.env.STRIPE_PRICE_ID_PRO, // R$ 9.99/mês
      metadata: { game: 'onzap' }
    });

    if (subscription.success) {
      // Mark as Pro user
      await this.prisma.user.update({
        where: { id: userId },
        data: { isPro: true }
      });

      // Give bonus ONZAP access
      await this.onzapService.grantProAccess(userId, 1); // 1 month

      return { success: true, subscription };
    }
  }
}
```
